<script type="text/javascript">
(function () {
  // Same reasoning as full.liquid.txt: the framework's own Fit Value
  // runtime doesn't appear to execute in TRMNL's actual render
  // pipeline, confirmed on device, so this view shrinks its own verdict
  // text rather than depending on it. data-value-fit left in place too,
  // harmless, in case it ever does fire.
  function shrinkToFit(el, minPx) {
    if (!el) return;
    var container = el.parentElement;
    if (!container) return;
    var guard = 0;
    while (el.scrollWidth > container.clientWidth + 1 && guard < 60) {
      var size = parseFloat(window.getComputedStyle(el).fontSize);
      if (!size || size <= minPx) break;
      el.style.fontSize = (size - 1) + "px";
      guard++;
    }
  }
  var els = document.querySelectorAll(".fit-text");
  for (var i = 0; i < els.length; i++) shrinkToFit(els[i], 14);
})();
</script>
