  name: Location

  optional: false

- keyname: location_name
  field_type: string
  name: Location name
  description: Shown on the screen. Leave empty to show the coordinates.
  optional: true

- keyname: use_real_map
  field_type: boolean
  name: Use real map
  description: Show a real OpenStreetMap tile map instead of the abstract polar chart.
  default: "false"
  optional: true

- keyname: time_format
  field_type: select
  name: Time format
  description: How clock times are shown (for example the best-viewing window).
  options:

    - "12h"
  default: "24h"
  optional: true

  field_type: select
  name: Voice
  description: How the verdict is worded. Numbers (times, Kp) stay the same either way.
  options:
    - facts
    - astrophysicist
  default: facts
  optional: true

- keyname: language
  field_type: select
  name: Language
  description: Screen language.
  options:

  optional: true
